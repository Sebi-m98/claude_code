Sub Sortieren_Klicken()
    Dim ws As Worksheet
    Set ws = ActiveSheet
    If LeerzeilenCheck.AbbrechenWennLeereZeilenVorhanden(ws) Then
        MsgBox "Sortierung nicht mehr möglich! Leerzeilen wurden aktiviert. Bitte deaktiviere die Leerzeilen, um die Funktion nutzen zu können.", vbExclamation, "Leerzeilen gefunden!"
        Exit Sub
    End If
    If ActiveWorkbook.Sheets("Schaltzettel").Cells(1, 1).Value = "" Then
        frmSortieren.Show
    Else
        MsgBox "Sortierung nicht mehr möglich, Umschaltung bereits durchgeführt."
    End If

End Sub

Sub ReplaceIdentifiersInCSV()
    Dim ws As Worksheet
    Dim filePath As String
    Dim fileContent As String
    Dim textLines As Variant
    Dim identifiers As Variant
    Dim i As Integer, j As Integer
    Dim firstCol As String
    Dim newFirstCol As String
    Dim separator As String
    Dim relpaceWith As String
    Dim response As VbMsgBoxResult
    
    identifiers = Array("H20", "H16", "BSL", "H02", "H14", "SVD", "DSJ", "RSJ", "RSD", "H03", "H11", "H04", "VVR", "VSL", "H15", "H18", "DSL", "H21", "ZGL", "H13", "H19", "RSL", "PSL", "SVR", "VVD", "VRL", "H05")
    separator = ";"
    replaceWith = "/-/"
    
    response = MsgBox("Bitte Beachte: Diese Funktion ist lediglich ein kurzfristiger Bugfix (Stand: März 2025) für die fehlerhaften Zusatzparameter in den Leitungsbezeichnungen." + vbCrLf + vbCrLf + "Bitte wähle im folgenden die exportierte CSV-Datei aus. Diese wird vom Makro automatisch bearbeitet und kann danach über den Button 'CSV Import' wieder eingelesen werden.", vbOKCancel + vbInformation)
    
    If response = vbCancel Then
        Exit Sub
    End If
    
    ' CSV auswählen
    With Application.FileDialog(msoFileDialogOpen)
        .AllowMultiSelect = False
        .Filters.Clear
        .Filters.Add "CSV Files", "*.csv"
        If .Show Then
            filePath = .SelectedItems(1)
        Else
            Exit Sub
        End If
    End With
    
    ' CSV einlesen
    Open filePath For Input As #1
    fileContent = Input$(LOF(1), #1)
    Close #1
    
    ' Zeilen durchlaufen und nach identifiers suchen und ersetzen
    textLines = Split(fileContent, vbLf)
    For i = LBound(textLines) To UBound(textLines)
        If textLines(i) <> "" Then
            Dim columns As Variant
            columns = Split(textLines(i), separator)
            firstCol = columns(0)
            
            For j = LBound(identifiers) To UBound(identifiers)
                newFirstCol = Replace(firstCol, "/" & identifiers(j) & "/", replaceWith)
                If newFirstCol <> firstCol Then

                End If
            Next j
            
            columns(0) = firstCol
            textLines(i) = Join(columns, separator)
        End If
    Next i
    
    ' Modifiziert CSV speichern
    Open filePath For Output As #1
    Print #1, Join(textLines, vbCrLf)
    Close #1
    
    MsgBox "Die CSV-Datei wurde erfolgreich geändert! Die Datei kann nun über den Button 'CSV Import' eingelesen werden!", vbInformation

Sub LeerzeilenEinfuegenOderLoeschen()
    Dim ws As Worksheet
    Set ws = ActiveSheet

    If LeerzeilenCheck.AbbrechenWennLeereZeilenVorhanden(ws) Then
        ' Es wurde in Spalte C "Leerzeile" gefunden, also wird der Löschprozess gestartet
        Dim confirmDelete As VbMsgBoxResult
        confirmDelete = MsgBox( _
            "Es wurden Leerzeilen gefunden. Diese Zeilen werden jetzt gelöscht und alle Inhalte darin gehen verloren!" & vbCrLf & vbCrLf & _
            "Möchtest du fortfahren?", _
            vbExclamation + vbYesNo, "Löschen bestätigen")

        If confirmDelete = vbYes Then
            Call LeerzeilenEntfernen
        Else
            MsgBox "Löschen abgebrochen.", vbInformation
        End If
    Else
        ' Leerzeile nicht in Spalte C gefunden, also wird der Einfüge-Prozess gestartet
        Call LeerzeilenEinfuegen
    End If
End Sub


Sub LeerzeilenEinfuegen()
    Dim ws As Worksheet
    Dim lastRow As Long
    Dim blockFirstRow As Long
    Dim blockRow As Long
    Dim blockLastRow As Long
    Dim insertedRow As Long

    Set ws = ActiveSheet
    Application.ScreenUpdating = False

    ' Bestimme letzte belegte Zeile in Spalte D (oder A, falls D leer ist)
    lastRow = ws.Cells(ws.Rows.Count, "D").End(xlUp).Row
    blockFirstRow = 3

    Do While blockFirstRow <= lastRow
        If ws.Cells(blockFirstRow, "D").MergeCells Then
            mergeHeight = ws.Cells(blockFirstRow, "D").MergeArea.Rows.Count
        Else
            mergeHeight = 1
        End If
    
        blockRow = 0
        Do While blockRow < mergeHeight
            insertedRow = blockFirstRow + (blockRow * 2) + 1
            ws.Rows(insertedRow).Insert Shift:=xlDown

            With ws.Range(ws.Cells(insertedRow, "F"), ws.Cells(insertedRow, "AC"))
                .Font.Color = RGB(255, 0, 0)
            End With

            With ws.Cells(insertedRow, "C")
                .Value = "Leerzeile"
                .Font.Italic = True
                .Font.Color = RGB(180, 180, 180)
            End With
            
            blockRow = blockRow + 1
        Loop
        
        blockLastRow = blockFirstRow + (mergeHeight * 2) - 1
        With ws.Range(ws.Cells(blockFirstRow, "D"), ws.Cells(blockLastRow, "D"))
            .Merge
            .VerticalAlignment = xlCenter
        End With
        With ws.Range(ws.Cells(blockFirstRow, "E"), ws.Cells(blockLastRow, "E"))
            .Merge
            .VerticalAlignment = xlCenter
        End With
        
        ' Alle inneren Rahmenlinien entfernen
        With ws.Range(ws.Cells(blockFirstRow, "D"), ws.Cells(blockLastRow, "AC"))
            .Borders(xlInsideHorizontal).LineStyle = xlLineStyleNone
            .Borders(xlInsideVertical).LineStyle = xlLineStyleNone
        End With

        blockFirstRow = blockLastRow + 1 ' neue erste Zeile für block finden
        lastRow = lastRow + mergeHeight ' weil "mergeHeight" neue Zeilen hinzugefügt wurde
    Loop

    Application.ScreenUpdating = True
    MsgBox "Leerzeilen wurden eingefügt.", vbInformation
End Sub

Sub LeerzeilenEntfernen()
    Dim ws As Worksheet
    Dim i As Long
    Dim lastRow As Long
    Set ws = ActiveSheet

    Application.ScreenUpdating = False

    ' Letzte Zeile in Spalte C
    lastRow = ws.Cells(ws.Rows.Count, "C").End(xlUp).Row

    ' Rückwärts durchgehen, um Löschung korrekt zu verarbeiten
    For i = lastRow To 3 Step -1
        If Trim(LCase(ws.Cells(i, "C").Value)) = "leerzeile" Then
            ws.Rows(i).Delete
        End If
    Next i

    Application.ScreenUpdating = True
    MsgBox "Alle Leerzeilen wurden entfernt.", vbInformation
End Sub

                ' Verbundene Zellen trennen und in allen Zellen den selben Wert schreiben
                For Each mrgCelle In Intersect(columns("D:E"), ActiveSheet.UsedRange)
                    If mrgCelle.MergeCells Then
                        With mrgCelle.MergeArea
                            .UnMerge
                            .Value = mrgCelle.Value
                        End With
                    End If
                Next mrgCelle
            End If
            
            ' und die Spalten-Nr. des HVT-L ermitteln
            ' nach einem Fehler einen zusätzlichen Spaltennamen mit angefügt, da Flexprod zum Teil HVt-L als Titel angegeben hat
            intColHVtL = 0
    
            For intCol = 1 To intMaxCol + cCol Step 1
                If InStr(1, ActiveWorkbook.Sheets("Schaltzettel").Cells(lAktRow, intCol).Value, "HVT-L") Or InStr(1, ActiveWorkbook.Sheets("Schaltzettel").Cells(lAktRow, intCol).Value, "HVt-L") Then
                    If intColHVtL = 0 Then
                        intColHVtL = intCol
                    End If
                End If
            Next intCol
            
            lAktRow = 3 ' aktuelle Zeile-Nr. im Sortierdurchlauf
            lStartRow = 3 ' Start-Zeile für den nächsten Sortierdurchlauf
            lBegLtg = 0
            lEndLtg = 0
            lngEinfRow = 0
            blnCheck = False
            
            With ActiveWorkbook.Sheets("Schaltzettel")
                ' äußere Schleife, Kopierschleife / Sortieren
                Do
                    lBegLtg = 0
                    lEndLtg = 0
                    lAktRow = lStartRow
                    lFirstRow = lStartRow - 1                    ' erste Zeilen-Nr. einer Leitung
                    
                    strHVtL = ""
                    strReihe = ""
                    strStift = ""
                    strLeiste = ""
                    
                    intReiheNr = 0
                    intReiheMin = 1000
                    intStiftNr = 0
                    intStiftMin = 1000
                    intLeisteNr = 0
                    intLeisteMin = 1000
                    
    ' ############# Schleife für HVt-L, Suchschleife für minimalen Wert
            
                    Do
                        ' Neue Leitungskey
                        If .Cells(lAktRow, cCol) <> .Cells(lFirstRow, cCol) Then
                            lFirstRow = lAktRow
                        End If
                    
                        strHVtL = .Cells(lAktRow, intColHVtL)
                        arrHVtL = Split(strHVtL, "/") ' String auflösen in HVt Reihe Leiste und Stift
                        
                        If UBound(arrHVtL) > 1 Then
                            strReihe = arrHVtL(0)
                            intReiheNr = Val(strReihe)
                            
                            ' prüfen, ob die Reihe kleiner wurde
                            If intReiheNr < intReiheMin Then
                            ' alle Min Werte setzen
                                strLeiste = arrHVtL(1)
                                strStift = arrHVtL(2)
                                
                                intLeisteNr = Val(strLeiste)
                                intStiftNr = Val(strStift)
                                
                                intReiheMin = intReiheNr
                                intLeisteMin = intLeisteNr
                                intStiftMin = intStiftNr
                                
                                lBegLtg = lFirstRow
                                
                                ' Prüfungbit, ob die innere Schleife mindestens einmal durchlaufen wird
                                blnCheck = True
    
                                ' Wenn gleich ein Wechsel der Leitung erfolgt
                                If .Cells(lAktRow + 1, cCol) <> .Cells(lBegLtg, cCol) Then
                                    lEndLtg = lAktRow
                                Else
                                
                                ' Wenn die Leitung gleich bleibt, bis zum Ende der Ltg. gehen
                                    Do While .Cells(lAktRow, cCol) = .Cells(lBegLtg, cCol)
                                        lAktRow = lAktRow + 1
                                    Loop
                                    lAktRow = lAktRow - 1
                                    lEndLtg = lAktRow
                                End If

                            Else
                                If intReiheNr = intReiheMin Then
                                    ' Prüfung Leistennr.
                                    strLeiste = arrHVtL(1)
                                    intLeisteNr = Val(strLeiste)
                                    
                                    ' prüfen, ob die Leiste kleiner wurde
                                    If intLeisteNr < intLeisteMin Then
                                        strStift = arrHVtL(2)
                                        intStiftNr = Val(strStift)
                                        intLeisteMin = intLeisteNr
                                        intStiftMin = intStiftNr
                                        
                                        lBegLtg = lFirstRow

                                        ' Wenn gleich ein Wechsel der Leitung erfolgt
                                       If .Cells(lAktRow + 1, cCol) <> .Cells(lBegLtg, cCol) Then
                                           lEndLtg = lAktRow
                                       Else
                                       
                                       ' Wenn die Leitung gleich bleibt, bis zum Ende der Ltg. gehen
                                           Do While .Cells(lAktRow, cCol) = .Cells(lBegLtg, cCol)
                                               lAktRow = lAktRow + 1
                                           Loop
                                           lAktRow = lAktRow - 1
                                           lEndLtg = lAktRow
                                       End If

                                    Else
                                        If intLeisteNr = intLeisteMin Then
                                            ' prüfen, ob Stiftnr. kleiner wurde
                                            strStift = arrHVtL(2)
                                            intStiftNr = Val(strStift)
                                            If intStiftNr < intStiftMin Then
                                                intStiftMin = intStiftNr
                                                lBegLtg = lFirstRow

                                                ' Wenn gleich ein Wechsel der Leitung erfolgt
                                                If .Cells(lAktRow + 1, cCol) <> .Cells(lBegLtg, cCol) Then
                                                    lEndLtg = lAktRow
                                                Else
                                                
                                                ' Wenn die Leitung gleich bleibt, bis zum Ende der Ltg. gehen
                                                    Do While .Cells(lAktRow, cCol) = .Cells(lBegLtg, cCol)
                                                        lAktRow = lAktRow + 1
                                                    Loop
                                                    lAktRow = lAktRow - 1
                                                    lEndLtg = lAktRow
                                                End If
                                            End If
                                        End If
                                    End If
                                End If
                            End If
                            
                        End If
                        
                        lAktRow = lAktRow + 1
                        
                    Loop While .Cells(lAktRow, cCol) <> ""
                    
                    ' Falls beim 1. Durchlauf keine Hvt-L Bezeichnung gefunden, Abbruch
                    If Not blnCheck Then
                        GoTo AbbruchHvtL
                    End If
                        
                    If lBegLtg > 0 And lEndLtg > 0 Then
                        ' Ltg mit minimaler Stiftnr. an Startposition einfügen
                        lngEinfRow = lEndLtg - lBegLtg + 1
                        For lngZahl = 1 To lngEinfRow
                            .Rows(lStartRow & ":" & lStartRow).Insert Shift:=xlDown
                        Next lngZahl
                        .Range(.Cells(lBegLtg + lngEinfRow, cCol), Cells(lEndLtg + lngEinfRow, intMaxCol)).Copy Destination:=.Cells(lStartRow, 4)
                        .Rows(lBegLtg + lngEinfRow & ":" & lEndLtg + lngEinfRow).Delete Shift:=xlUp
        
                    End If
                    
                    ' nach der Einfügezeile fortfahren und die nächste minimale Stiftnr. ermitteln
                    lStartRow = lStartRow + lngEinfRow
                    
                    DoEvents
                Loop While lStartRow < lngMaxRow
            
            End With
            
            strMsg = "Sortierung nach HVt-L"
        End If
        
    End If
    
' ################# jede zweite Leitung grauer Hintergrund ###############

    lAktRow = 3                            ' Gilt ab Zeile-Nr.
    lFirstRow = lAktRow
    lFirstLSZ = 3
    lFirstKey = 3
    wechsel = 0
    
    With ActiveSheet
        Application.DisplayAlerts = False
        Do
            lAktRow = lAktRow + 1
            If .Cells(lAktRow, cCol) <> .Cells(lFirstRow, cCol) Then
            
                .Range(.Cells(lAktRow - 1, cCol), .Cells(lFirstRow, intMaxCol)).Select ' intMaxCol bedeutet wie weit geht es nach rechts
            
                Selection.Borders(xlDiagonalDown).LineStyle = xlNone
                Selection.Borders(xlDiagonalUp).LineStyle = xlNone
                With Selection.Borders(xlEdgeLeft)
                    .LineStyle = xlContinuous
                    .ColorIndex = 0
                    .TintAndShade = 0
                    .Weight = xlThin
                End With
                With Selection.Borders(xlEdgeTop)
                    .LineStyle = xlContinuous
                    .ColorIndex = 0
                    .TintAndShade = 0
                    .Weight = xlThin
                End With
                With Selection.Borders(xlEdgeBottom)
                    .LineStyle = xlContinuous
                    .ColorIndex = 0
                    .TintAndShade = 0
                End With

                ' zuerst kein Hintergund
                .Range(.Cells(lAktRow - 1, cCol), .Cells(lFirstRow, intMaxCol)).Select ' intMaxCol bedeutet wie weit geht es nach rechts
                With Selection.Interior
                    .Pattern = xlNone
                    .TintAndShade = 0
                    .PatternTintAndShade = 0
                End With
                If wechsel Mod 2 <> 0 Then
                    ' jede zweite Ltg grauer Hintergrund
                    With Selection.Interior
                        .Pattern = xlSolid
                        .PatternColorIndex = xlAutomatic
                        .ThemeColor = xlThemeColorDark1
                        .TintAndShade = -0.149998474074526
                        .PatternTintAndShade = 0
                    End With
                End If
                
                ' gleiche LSZ Verbinden
                .Range(.Cells(lAktRow - 1, cCol + 1), .Cells(lFirstLSZ, cCol + 1)).Merge (False) 'weitere verbundene Zellen
                
                ' gleiche Leitungskey verbinden
                If intMaxCol > 29 Then  ' großer Schaltzettel
                    .Range(.Cells(lAktRow - 1, cCol + 2), .Cells(lFirstKey, cCol + 2)).Merge (False) 'weitere verbundene Zellen
                End If

                .Range(.Cells(lAktRow - 1, cCol), .Cells(lFirstRow, cCol)).Merge (False)
                
                lFirstLSZ = lAktRow
                lFirstKey = lAktRow

                lFirstRow = lAktRow
' ###Zähler für gerade/ungerade Überprüfung um 1 erhöhen
                wechsel = wechsel + 1
            Else
                    
                ' Leitungsbez bebleibt gleich, nur LSZ bzw. Key ändert sich
                ' Wechsel der LSZ  bzw. Leitungskey bei gleicher Leitungsbez
                If .Cells(lAktRow, cCol + 1) <> .Cells(lFirstLSZ, cCol + 1) Then
                    .Range(.Cells(lAktRow - 1, cCol + 1), .Cells(lFirstLSZ, cCol + 1)).Merge (False) 'weitere verbundene Zellen
                    lFirstLSZ = lAktRow
                End If
                
                ' Wechsel des Leitungskey bei gleicher Leitungsbez
                If intMaxCol > 29 Then  ' großer Schaltzettel
                    If .Cells(lAktRow, cCol + 2) <> .Cells(lFirstKey, cCol + 2) Then
                        .Range(.Cells(lAktRow - 1, cCol + 2), .Cells(lFirstKey, cCol + 2)).Merge (False) 'weitere verbundene Zellen
                        lFirstKey = lAktRow
                    End If
                End If

            
            End If
        Loop While .Cells(lAktRow, cCol) <> ""
        .Range("D3").Select
        Application.DisplayAlerts = True
    End With
        

    Application.DisplayAlerts = True
    Application.ScreenUpdating = True 'Bildschirmaktualisierung an

    MsgBox strMsg & " mit Erfolg beendet!"

Exit Sub

'  ############## Abbrueche ##############
Abbruch:
    MsgBox "Angegebene EVS Nr. zum Sortieren nicht gefunden."
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True 'Bildschirmaktualisierung an
    Exit Sub
    
AbbruchAPL:
    MsgBox "Fehler bei der APL Sortierung."
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True 'Bildschirmaktualisierung an
    Exit Sub
    
AbbruchHvtL:
    MsgBox "Keine Einträge bei der Hvt-L Sortierung gefunden."
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True 'Bildschirmaktualisierung an
    Exit Sub
    
' ############ Fehlerausgabe
Ende:
    Application.ScreenUpdating = True 'Bildschirmaktualisierung an
    MsgBox Error.Description
    Exit Sub

End Sub

        For Each mrgCelle In Intersect(columns("D:E"), ActiveSheet.UsedRange)
            If mrgCelle.MergeCells Then
                With mrgCelle.MergeArea
                    .UnMerge
                    .Value = mrgCelle.Value
                End With
            End If
        Next mrgCelle
            
        If Me.txtStiftNrBeg <> "" Then
            intStiftNeu = CInt(Me.txtStiftNrBeg)
        End If
        
        If Me.txtKVzNr2 <> "" And IsNumeric(Me.txtKVzNr2) = False Then

            For intCol = 1 To intMaxCol Step 1

                    End If

                If UBound(arrEVSStiftAb) > 0 Then

                            End If
                        
                        Else

                        End If
                    Else

                    End If
                Else

                End If

            End If
                
UmschaltungJa:

                   ActiveWorkbook.Sheets("Schaltzettel").Cells(lngRow, intColEVSStiftAb).FormulaR1C1 = strEVSab & Me.txtEVSneu & "/" & strStiftab & intStiftNeu

                   ActiveWorkbook.Sheets("Schaltzettel").Cells(lngRow, intColEVSStiftAb).FormulaR1C1 = strEVSab & Me.txtEVSneu & "/" & strStiftab

                    .Shadow = False

                        .Strikethrough = False
                        .Superscript = False
                        .Subscript = False
                        .OutlineFont = False
                        .Shadow = False
                        .Underline = xlUnderlineStyleNone

                    End With
                
                    intStiftNeu = intStiftNeu + 1
                End If
            End If
            DoEvents
            intAnzahl = intAnzahl + 1
            
UmschaltungNein:

        Next lngRow
        
'        ActiveWorkbook.Sheets("Schaltzettel").Cells(1, 1).FormulaR1C1 = 1
    Else
        MsgBox "Format von Stift neu Beginnwert nicht korrekt."
        Exit Sub
    End If
    
' Spalte auf die optimale Breite bringen
    With ActiveWorkbook.Sheets("Schaltzettel")
        .columns(intColEVSStiftAb).EntireColumn.AutoFit
    End With
    
    ' Gleiche Leitungsbezeichnungen und LSZ wieder verbinden
    Call ZellenVerbinden
    
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True 'Bildschirmaktualisierung an

    MsgBox "Umschaltung auf 'Stift ab' bei " & intAnzahl & " Leitungen durchgeführt."
'    Me.cmdUmschEvsAb.Visible = False
    Exit Sub
    
Ende:
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True 'Bildschirmaktualisierung an

    MsgBox "Umschaltung nicht vollständig erfolgt, bitte mit neuen Daten durchführen!"
    Exit Sub

End Sub

    Dim lngMaxRow, lngRow As Long ' maximale Zeilen der Schaltzetteltabelle, Zeilen-Zähler

        strAsbKvzAlt = ActiveWorkbook.Sheets("Schaltzettel").Cells(lngRow, intColKVz).Value
'        MsgBox strAsbKvzAlt
        
        If InStr(strAsbKvzAlt, "/") > 1 Then
            arrAsbKvzAlt = Split(strAsbKvzAlt, "/") ' String auflösen in Reihe, Leiste und Stift
            If UBound(arrAsbKvzAlt) > 0 Then
                strAsb = arrAsbKvzAlt(LBound(arrAsbKvzAlt))
                strKvzAlt = arrAsbKvzAlt(UBound(arrAsbKvzAlt))
                If strKvzAlt = Me.txtKVzAlt Then
                    GoTo UmschaltungJa
                Else
                ' anderer KVz
                    GoTo UmschaltungNein
                End If
            Else
                GoTo UmschaltungNein
            End If
        Else
            ' keine Umschaltung, da keine KVzbezeichnung
            GoTo UmschaltungNein
        End If
            
UmschaltungJa:
        intLaengeAsb = Len(strAsb)
        intLaengeKvzAlt = Len(strKvzAlt)
'        MsgBox strAsb & "/" & strKvzAlt & " " & strKvzNeu
        ActiveWorkbook.Sheets("Schaltzettel").Cells(lngRow, intColKVz).FormulaR1C1 = strAsb & "/" & strKvzAlt & "" & strKvzNeu
        
'       KVz alt durchgestrichen
        With ActiveWorkbook.Sheets("Schaltzettel").Cells(lngRow, intColKVz).Characters(Start:=intLaengeAsb + 2, Length:=intLaengeKvzAlt).Font
            .Name = "Calibri"
            .FontStyle = "Standard"
            .Size = 11
            .Strikethrough = True
            .Superscript = False
            .Subscript = False
            .OutlineFont = False
            .Shadow = False
            .Underline = xlUnderlineStyleNone
            .ThemeColor = xlThemeColorLight1
            .TintAndShade = 0
            .ThemeFont = xlThemeFontMinor
        End With

'       KVz neu rot markiert
        With ActiveWorkbook.Sheets("Schaltzettel").Cells(lngRow, intColKVz).Characters(Start:=intLaengeAsb + 2 + intLaengeKvzAlt, Length:=intLaengeKvzNeu).Font
            .Name = "Calibri"
            .FontStyle = "Standard"
            .Size = 11
            .Strikethrough = False
            .Superscript = False
            .Subscript = False
            .OutlineFont = False
            .Shadow = False
            .Underline = xlUnderlineStyleNone
            .Color = -16776961
            .TintAndShade = 0
            .ThemeFont = xlThemeFontMinor
        End With



        intAnzahl = intAnzahl + 1
            
UmschaltungNein:
        DoEvents
            
    Next lngRow
    
' Spalte auf die optimale Breite bringen
    With ActiveWorkbook.Sheets("Schaltzettel")
        .columns(intColKVz).EntireColumn.AutoFit
    End With
    
    ' Gleiche Leitungsbezeichnungen und LSZ wieder verbinden
    Call ZellenVerbinden
    
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True 'Bildschirmaktualisierung an

    MsgBox "Umschaltung von 'KVz' bei " & intAnzahl & " Leitungen durchgeführt."
'    Me.cmdUmschHVTL.Visible = False
    Exit Sub
    
Ende:
    Application.DisplayAlerts = True
    Application.ScreenUpdating = True 'Bildschirmaktualisierung an

    MsgBox "Umschaltung nicht vollständig erfolgt, bitte mit neuen Daten durchführen!"
    Exit Sub
    
End Sub

Private Sub ZellenVerbinden()
    Dim lMaxRow, lAktRow, lFirstRow, lFirstLSZ As Long
    Const cCol As Long = 4              ' Gilt für Spalte-Nr. Leitungsbezeichnung
    lAktRow = 3                            ' Gilt ab Zeile-Nr.
    lFirstRow = lAktRow
    lFirstLSZ = 3
    lMaxRow = ActiveWorkbook.Sheets("Schaltzettel").UsedRange.Rows.Count

    With ActiveSheet
        Application.DisplayAlerts = False
        Do
            lAktRow = lAktRow + 1
            If .Cells(lAktRow, cCol) <> .Cells(lFirstRow, cCol) Then
                ' gleiche Leitungsbezeichnung und LSZ Verbinden
                .Range(.Cells(lAktRow - 1, cCol), .Cells(lFirstRow, cCol)).Merge (False)
                .Range(.Cells(lAktRow - 1, cCol + 1), .Cells(lFirstLSZ, cCol + 1)).Merge (False) 'weitere verbundene Zellen
                lFirstRow = lAktRow
                lFirstLSZ = lAktRow
            Else
                If .Cells(lAktRow, cCol + 1) <> .Cells(lFirstLSZ, cCol + 1) Then
                    ' Lreitungsbezeichnung bleibt gleich, nur LSZ Wechsel und gleiche Zellen verbinden
                    .Range(.Cells(lAktRow - 1, cCol + 1), .Cells(lFirstLSZ, cCol + 1)).Merge (False) 'weitere verbundene Zellen
                    lFirstLSZ = lAktRow
                End If

            End If
        Loop While .Cells(lAktRow, cCol) <> "" ' lAktRow <= lMaxRow + 1
            
    End With
End Sub

        For Each mrgCelle In Intersect(columns("D:F"), ActiveSheet.UsedRange)

                    .UnMerge

                    .UnMerge
                    .Value = mrgCelle.Value
                End With
            End If
        Next mrgCelle
    End If
    DoEvents

' Spalte für LSZ suchen, hier ist zurue Angabe
    For intCol = 1 To intMaxCol + cCol Step 1
        If InStr(1, ActiveWorkbook.Sheets("Schaltzettel").Cells(lAktRow, intCol).Value, "LSZ") Then

    lAktRow = 3                            ' Gilt ab Zeile-Nr.
    lFirstRow = lAktRow
    lFirstLSZ = 3
    lFirstKey = 3
    wechsel = 0
    
    With ActiveSheet
        Application.DisplayAlerts = False
        Do
            lAktRow = lAktRow + 1
            If .Cells(lAktRow, cCol) <> .Cells(lFirstRow, cCol) Then
            
                .Range(.Cells(lAktRow - 1, cCol), .Cells(lFirstRow, intMaxCol)).Select ' intMaxCol bedeutet wie weit geht es nach rechts
            
                Selection.Borders(xlDiagonalDown).LineStyle = xlNone
                Selection.Borders(xlDiagonalUp).LineStyle = xlNone
                With Selection.Borders(xlEdgeLeft)
                    .LineStyle = xlContinuous

                End With
                With Selection.Borders(xlEdgeTop)

                With Selection.Borders(xlEdgeBottom)

                    .ColorIndex = 0

                .Range(.Cells(lAktRow - 1, cCol), .Cells(lFirstRow, intMaxCol)).Select ' intMaxCol bedeutet wie weit geht es nach rechts
                With Selection.Interior
                    .Pattern = xlNone
                    .TintAndShade = 0

                lFirstLSZ = lAktRow
                lFirstKey = lAktRow
                
                lFirstRow = lAktRow

                wechsel = wechsel + 1
            Else
            ' Rufnummer bebleibt gleich, nur LSZ bzw. Key ändert sich
            ' Wechsel der LSZ  bzw. Leitungskey bei gleicher Rufnummer
                If .Cells(lAktRow, cCol + 1) <> .Cells(lFirstLSZ, cCol + 1) Then
                    .Range(.Cells(lAktRow - 1, cCol + 1), .Cells(lFirstLSZ, cCol + 1)).Merge (False) 'weitere verbundene Zellen
                    lFirstLSZ = lAktRow

                    If .Cells(lAktRow, cCol + 2) <> .Cells(lFirstKey, cCol + 2) Then
                        .Range(.Cells(lAktRow - 1, cCol + 2), .Cells(lFirstKey, cCol + 2)).Merge (False) 'weitere verbundene Zellen
                        lFirstKey = lAktRow
                    End If
                End If
            End If
            DoEvents
            
        Loop While .Cells(lAktRow, cCol) <> ""
        .Range("D3").Select
        Application.DisplayAlerts = True
    End With
    
    Application.ScreenUpdating = True
    MsgBox "" & lngZahl & " Zeilen gelöscht."

End Sub

Function func_MaxColumn(lngRow As Long) As Integer
    Dim intColAkt As Integer
    intColAkt = 1
    
    With ActiveWorkbook.Sheets("Schaltzettel")

                func_MaxColumn = intColAkt - 1

            End If
        Next intColAkt
    End With
End Function

Function func_MaxRow(intCol As Integer) As Long

    With ActiveWorkbook.Sheets("Schaltzettel")
    
        For lngRowAkt = 3 To 10000

                func_MaxRow = lngRowAkt - 1
                Exit Function
            End If
        Next lngRowAkt
    End With
End Function

Function AbbrechenWennLeereZeilenVorhanden(ws As Worksheet) As Boolean
    Dim i As Long
    Dim lastRow As Long

    lastRow = ws.Cells(ws.Rows.Count, "C").End(xlUp).Row

    For i = 3 To lastRow
        If Trim(LCase(ws.Cells(i, "C").Value)) = "leerzeile" Then
            AbbrechenWennLeereZeilenVorhanden = True
            Exit Function
        End If
    Next i

    AbbrechenWennLeereZeilenVorhanden = False
End Function
